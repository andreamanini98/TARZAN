create arena MedApp
{
    clocks
    {
        x, y, t;
    }

    actions
    {
        draw,
        prepare,
        restore,
        arrive,
        handoff,
        ack_hand,
        check,
        ack,
        ack_safe,
        ack_nurse,
        interrupt,
        retry,
        call_nurse,
        wait_nurse,
        dispose,
        reset,
        to_dock;
    }

    locations
    {
        q_dock      <player: c, <ini: T>>,
        q_draw      <player: c, <ini: F, inv: [(x, <=, 3)]>>,
        q_exp       <player: e, <ini: F>>,
        q_prep      <player: c, <ini: F>>,
        q_room      <player: c, <ini: F>>,
        q_hand      <player: e, <ini: F>>,
        q_pulse     <player: c, <ini: F>>,
        q_int       <player: e, <ini: F>>,
        qD_pulse    <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD1_safe    <player: e, <ini: F>>,
        q_blood     <player: c, <ini: F>>,
        qD_blood    <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD2_safe    <player: e, <ini: F>>,
        q_drug      <player: c, <ini: F>>,
        qD_drug     <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD3_safe    <player: e, <ini: F>>,
        q_disp      <player: c, <ini: F>>,
        q_reset     <player: c, <ini: F>>,
        q_back      <player: c, <ini: F>>,
        q_call      <player: c, <ini: F>>,
        q_nurse     <player: e, <ini: F>>,
        qN_pulse    <player: c, <ini: F>>,
        qN_int      <player: e, <ini: F>>,
        qDN_pulse   <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD1N_safe   <player: e, <ini: F>>,
        qN_blood    <player: c, <ini: F>>,
        qDN_blood   <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD2N_safe   <player: e, <ini: F>>,
        qN_drug     <player: c, <ini: F>>,
        qDN_drug    <player: e, <ini: F, inv: [(y, <=, 6)]>>,
        qD3N_safe   <player: e, <ini: F>>;
    }

    transitions
    {
        (q_dock,    draw,       [(x, ==, 1)], [x],    q_draw),
        (q_draw,    prepare,    [(x, ==, 1)], [x],    q_prep),
        (q_draw,    prepare,    [(x, ==, 2)], [],     q_exp),
        (q_exp,     restore,    [(y, ==, 5)], [x, y], q_dock),
        (q_prep,    arrive,     [(x, ==, 2)], [x, y], q_room),
        (q_room,    handoff,    [(x, ==, 2)], [],     q_hand),
        (q_hand,    ack_hand,   [(y, ==, 3)], [x, y], q_pulse),
        (q_hand,    ack_hand,   [(y, ==, 1)], [],     q_exp),
        (q_int,     retry,      [(y, ==, 2)], [x, y], q_pulse),

        (q_pulse,   check,      [(x, ==, 2)], [],     qD_pulse),
        (qD_pulse,  ack,        [(y, ==, 3)], [x, y], q_blood),
        (qD_pulse,  ack,        [(y, ==, 3)], [y],    qD1_safe),
        (qD_pulse,  interrupt,  [(y, ==, 1)], [],     q_int),
        (qD_pulse,  interrupt,  [(y, ==, 1)], [x],    q_call),
        (qD1_safe,  ack_safe,   [(y, ==, 2)], [x, y], q_blood),

        (q_blood,   check,      [(x, ==, 2)], [],     qD_blood),
        (qD_blood,  ack,        [(y, ==, 3)], [x, y], q_drug),
        (qD_blood,  ack,        [(y, ==, 3)], [y],    qD2_safe),
        (qD_blood,  interrupt,  [(y, ==, 1)], [],     q_int),
        (qD_blood,  interrupt,  [(y, ==, 1)], [x],    q_call),
        (qD2_safe,  ack_safe,   [(y, ==, 2)], [x, y], q_drug),

        (q_drug,    check,      [(x, ==, 2)], [],     qD_drug),
        (qD_drug,   ack,        [(y, ==, 3)], [x],    q_disp),
        (qD_drug,   ack,        [(y, ==, 3)], [y],    qD3_safe),
        (qD_drug,   interrupt,  [(y, ==, 1)], [],     q_int),
        (qD_drug,   interrupt,  [(y, ==, 1)], [x],    q_call),
        (qD3_safe,  ack_safe,   [(y, ==, 2)], [x],    q_disp),

        (q_disp,    dispose,    [(x, ==, 2)], [x],    q_reset),
        (q_reset,   reset,      [(x, ==, 1)], [x],    q_back),
        (q_back,    to_dock,    [(x, ==, 2)], [x],    q_dock),

        (q_call,    call_nurse, [(x, ==, 1)], [x],    q_nurse),
        (q_nurse,   wait_nurse, [(y, ==, 3)], [x, y], qN_pulse),
        (qN_int,    retry,      [(y, ==, 2)], [x, y], qN_pulse),

        (qN_pulse,  check,      [(x, ==, 2)], [],     qDN_pulse),
        (qDN_pulse, ack,        [(y, ==, 3)], [x, y], qN_blood),
        (qDN_pulse, ack_nurse,  [(y, ==, 3)], [y],    qD1N_safe),
        (qDN_pulse, interrupt,  [(y, ==, 1)], [],     qN_int),
        (qD1N_safe, ack_safe,   [(y, ==, 2)], [x, y], qN_blood),

        (qN_blood,  check,      [(x, ==, 2)], [],     qDN_blood),
        (qDN_blood, ack,        [(y, ==, 3)], [x, y], qN_drug),
        (qDN_blood, ack_nurse,  [(y, ==, 3)], [y],    qD2N_safe),
        (qDN_blood, interrupt,  [(y, ==, 1)], [],     qN_int),
        (qD2N_safe, ack_safe,   [(y, ==, 2)], [x, y], qN_drug),

        (qN_drug,   check,      [(x, ==, 5)], [],     qDN_drug),
        (qDN_drug,  ack,        [(y, ==, 8)], [x],    q_disp),
        (qDN_drug,  ack_nurse,  [(y, ==, 3)], [y],    qD3N_safe),
        (qDN_drug,  interrupt,  [(y, ==, 1)], [],     qN_int),
        (qD3N_safe, ack_safe,   [(y, ==, 2)], [x],    q_disp);
    }
}
