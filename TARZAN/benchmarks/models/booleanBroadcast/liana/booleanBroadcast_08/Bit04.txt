create automaton Bit04
{
    clocks
    {
        x;
    }

    actions
    {
        tau, flip;
    }

    integers
    {
        ctr04;
    }

    locations
    {
        s0 <ini: T>,
        c0 <ini: F, com: T>,
        s1 <ini: F>,
        c1 <ini: F, com: T>;
    }

    transitions
    {
        (s0, tau, [(x, ==, 4)], [x], c0),
        (c0, flip!!, [], [], [ctr04 = 1 - ctr04], s1),
        (s1, tau, [(x, ==, 4)], [x], c1),
        (c1, flip!!, [], [], [ctr04 = 1 - ctr04], s0);
    }
}
