create arena production_cell
{
    clocks
    {
        x, y;
    }

    actions
    {
        toTable, takePlate_p, toPress, restore_p, takePlate_d, toDepot, restore_d, newPlate, newPlate_r;
    }

    locations
    {
        feed    <player: c, <ini: T, inv: [(x, <=, 2)]>>,
        table   <player: c, <ini: F>>,
        robot_p <player: c, <ini: F>>,
        fall_p  <player: e, <ini: F>>,
        press   <player: c, <ini: F>>,
        robot_d <player: c, <ini: F>>,
        fall_d  <player: e, <ini: F>>,
        depot   <player: c, <ini: F>>;
    }

    transitions
    {
        (feed,    toTable,     [(x, >=, 2)],             [x], table),
        (table,   takePlate_p, [(x, >=, 1), (x, <=, 3)], [x], robot_p),
        (robot_p, toPress,     [(x, <, 2)],              [x], fall_p),
        (robot_p, toPress,     [(x, >, 4)],              [x], fall_p),
        (fall_p,  restore_p,   [(x, >, 1), (x, <, 2)],   [],  feed),
        (robot_p, toPress,     [(x, <=, 4)],             [x], press),
        (press,   takePlate_d, [(x, >=, 3), (x, <=, 5)], [x], robot_d),
        (robot_d, toDepot,     [(x, <, 3)],              [x], fall_d),
        (robot_d, toDepot,     [(x, >=, 4)],             [x], fall_d),
        (fall_d,  restore_d,   [(x, >, 1), (x, <=, 5)],  [],  feed),
        (robot_d, toDepot,     [(x, >=, 3), (x, <=, 6)], [x], depot),
        (depot,   newPlate_r,  [],                       [x,y], feed),
        (depot,   newPlate,    [],                       [x], feed);
    }
}
