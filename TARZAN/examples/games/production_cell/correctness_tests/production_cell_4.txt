create arena production_cell
{
    clocks
    {
        x, y, zeno;
    }

    actions
    {
        toTable, takePlate_p, toPress, restore_p, takePlate_d, toDepot, restore_d, newPlate;
    }

    locations
    {
        feed    <player: c, <ini: T>>,
        table   <player: c, <ini: F>>,
        robot_p <player: c, <ini: F>>,
        fall_p  <player: e, <ini: F>>,
        press   <player: c, <ini: F>>,
        robot_d <player: c, <ini: F>>,
        fall_d  <player: e, <ini: F>>,
        depot   <player: c, <ini: F>>;
    }

    transitions
    {
        (feed,    toTable,     [(zeno, >, 0), (x, ==, 2)],             [zeno, x], table),
        (table,   takePlate_p, [(zeno, >, 0), (x, >=, 1), (x, <=, 3)], [zeno, x], robot_p),
        (robot_p, toPress,     [(zeno, >, 0), (x, <, 2)],              [zeno, x], fall_p),
        (robot_p, toPress,     [(zeno, >, 0), (x, >, 4)],              [zeno, x], fall_p),
        (fall_p,  restore_p,   [(zeno, >, 0), (x, >, 1), (x, <, 3)],   [zeno],  feed),
        (robot_p, toPress,     [(zeno, >, 0), (x, >=, 2), (x, <=, 4)], [zeno, x], press),
        (press,   takePlate_d, [(zeno, >, 0), (x, >=, 3), (x, <=, 5)], [zeno, x], robot_d),
        (robot_d, toDepot,     [(zeno, >, 0), (x, <, 3)],              [zeno, x], fall_d),
        (robot_d, toDepot,     [(zeno, >, 0), (x, >=, 4)],             [zeno, x], fall_d),
        (fall_d,  restore_d,   [(zeno, >, 0), (x, >, 1), (x, <, 2)],  [zeno],  feed),
        (robot_d, toDepot,     [(zeno, >, 0), (x, >=, 3), (x, <=, 6)], [zeno, x], depot),
        (depot,   newPlate,    [(zeno, >, 0) ],                        [zeno, x], feed);
    }
}
